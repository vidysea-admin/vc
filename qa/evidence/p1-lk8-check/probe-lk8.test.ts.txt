for (const [n,t] of pos) test("POS "+n, () => {
  const r = crisisGuard(t); assert.equal(r.level,"crisis",JSON.stringify(t)+" -> "+JSON.stringify(r));
  const g = routeBeforeGeneration(t); assert.equal(g.route,"escalate");
  if (g.route==="escalate") assert.equal(g.card.level,"crisis");
});
for (const [n,t] of neg) test("NEG "+n, () => {
  const r = crisisGuard(t); assert.equal(r.level,"none",JSON.stringify(t)+" -> "+JSON.stringify(r));
  assert.equal(routeBeforeGeneration(t).route,"continue");
});
