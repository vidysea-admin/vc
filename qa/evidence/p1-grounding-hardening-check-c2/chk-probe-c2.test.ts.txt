const NO_HITS = { status: "ok" as const, hits: [], dropped: 0 };
const NOT_SERVED = { status: "no_sourced_answer" as const, answer: NO_SOURCED_ANSWER_TEXT, citations: { internal: [], web: [] }, verdict: "incorrect" as const, reason: "", withheld: ["verdict_not_correct", "insufficient_coverage"] };
const noSourceKb = (anonymous = false) => fakeKb({ search: NO_HITS, ...(anonymous ? { anonymous: true } : { ask: NOT_SERVED as never }) });
const guidanceProvider = () => new MockProvider({ responses: { "kb-general-guidance": { answer: "Public universities in Germany usually charge no tuition." } as never } });
const groundedOk = (): GroundedAnswer => ({ text: "Most German public universities charge no tuition.", sources: [{ title: "daad.de", url: "https://vertexaisearch.cloud.google.com/grounding-api-redirect/AAA" }] });
const AUTH = { role: "student" as const, studentId: "syn-stu-3", tenantId: "sandbox" };
const turn = (text: string) => ({ mode: "authenticated" as const, auth: AUTH, threadId: "th-g", turnId: "tu-g", text });
function deps(over: Partial<StudentGraphDeps> = {}) {
  return { kb: fakeKb(), provider: studentProvider(), skillsDir: SKILLS_DIR, speakers: SPEAKERS, now: () => NOW, hasPriorFlag: NO_PRIOR_FLAG,
    recordGap: async () => {}, countAnonymous: async () => {}, onWriteError: () => {}, ...over } as StudentGraphDeps;
}
async function wire(text: string, anon = false): Promise<string | null> {
  const grounded = new MockGroundedAnswerer(groundedOk());
  await runStudentTurn(deps({ kb: noSourceKb(anon), provider: guidanceProvider(), grounding: grounded }), anon ? { mode: "anonymous", text } : turn(text));
  return grounded.calls.length ? grounded.calls[0]!.question : null;
}

test("CHK C2 direct redact", () => {
  const fails: string[] = [];
  for (const [t, toks] of MUST_REDACT) {
    const o = redactStudentDetails(t);
    const leaked = toks.filter((k) => o.toLowerCase().includes(k.toLowerCase()));
    console.log(`DIRECT|${leaked.length ? "LEAK" : "ok  "}|${JSON.stringify(t)} => ${JSON.stringify(o)}`);
    if (leaked.length) fails.push(t);
  }
  for (const [t, toks] of MUST_KEEP) {
    const o = redactStudentDetails(t);
    const lost = toks.filter((k) => !o.includes(k));
    console.log(`KEEP|${lost.length ? "LOST" : "ok  "}|${JSON.stringify(t)} => ${JSON.stringify(o)}`);
  }
  console.log("DIRECT_LEAK_COUNT=" + fails.length);
});

test("CHK C2 wire", async () => {
  let leaks = 0;
  for (const [t, toks] of MUST_REDACT) {
    for (const anon of [false]) {
      const q = await wire(t, anon);
      if (q === null) { console.log(`WIRE|nocall|${JSON.stringify(t)}`); continue; }
      const leaked = toks.filter((k) => q.toLowerCase().includes(k.toLowerCase()));
      console.log(`WIRE|${leaked.length ? "LEAK" : "ok  "}|${JSON.stringify(t)} => ${JSON.stringify(q)}`);
      if (leaked.length) leaks++;
    }
  }
  for (const [t, toks] of MUST_KEEP) {
    const q = await wire("german fees question: " + t);
    const lost = q === null ? ["nocall"] : toks.filter((k) => !q.includes(k));
    console.log(`WIREKEEP|${lost.length ? "LOST" : "ok  "}|${JSON.stringify(t)} => ${JSON.stringify(q)}`);
  }
  // anonymous path wire for the headline cases
  for (const t of ["I'm riya sharma from pune 411001. Which German universities suit PCM?", "my pin number is 411001, german fees?"]) {
    const q = await wire(t, true);
    console.log(`WIREANON|${q === null ? "nocall" : /411001|riya|sharma/i.test(q) ? "LEAK" : "ok  "}|${JSON.stringify(t)} => ${JSON.stringify(q)}`);
  }
  console.log("WIRE_LEAK_COUNT=" + leaks);
});
