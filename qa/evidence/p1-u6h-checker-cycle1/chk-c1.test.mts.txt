async function served(claim: string, evidence: string, kindc: "fact" | "number" = "fact"): Promise<boolean> {
  const provider = studentProvider({ kbAnswer: { claims: [{ text: claim, kind: kindc, sourceIds: ["t-x"] }] } });
  const d = { kb: fakeKb({ search: { status: "ok", hits: [hit(evidence)], dropped: 0 } }), provider, skillsDir: SKILLS_DIR, speakers: SPEAKERS, now: () => NOW, hasPriorFlag: NO_PRIOR_FLAG };
  const out = await runStudentTurn(d, turn("Tell me about the fees and rules for studying abroad please"));
  return out.response.claims.some((c: { status: string }) => c.status === "verified");
}

test("B. number-country binding (outcome)", async () => {
  const ev = "The contribution is 300 euros for Germany and 500 euros for France.";
  const bad: [string, string][] = [
    ["The contribution is 500 euros for Germany.", ev],
    ["The contribution in Germany is 500 euros.", ev],
    ["Germany charges 500 euros.", "Germany charges 300 euros and France charges 500 euros."],
    ["France charges 300 euros.", "Germany: 300 euros; France: 500 euros."],
    ["France charges 300 euros.", "Germany (300 euros) and France (500 euros)."],
    ["Germany charges 500 euros.", "Germany - 300 euros, France - 500 euros."],
    ["Germany charges 500 euros.", "300 euros in Germany, 500 in France."],
    ["Germany charges 500 euros.", "In Germany it is 300 euros, while in France it is 500 euros."],
    ["Germany charges 500 euros.", "France 500 euros, Germany 300 euros."],
    ["Germany charges 500 euros.", "Fees: Germany 300 euros / France 500 euros."],
  ];
  const leaks: string[] = [];
  for (const [c, e] of bad) if (await served(c, e, "number")) leaks.push(`${c} <- ${e}`);
  note(`B binding leaks: ${leaks.length}/${bad.length} ${JSON.stringify(leaks)}`);
  assert.equal(await served("The contribution is 300 euros for Germany.", ev, "number"), true);
  assert.equal(await served("The contribution is 500 euros for France.", ev, "number"), true);
  // first 3 are named by the repair
  assert.equal(await served("The contribution is 500 euros for Germany.", ev, "number"), false);
});

test("E. hasPriorFlag required; store is authoritative", async () => {
  const base = { kb: fakeKb(), provider: studentProvider(), skillsDir: SKILLS_DIR, speakers: SPEAKERS, now: () => NOW };
  assert.throws(() => buildStudentGraph(base as never), /hasPriorFlag/);
  await assert.rejects(runStudentTurn(base as never, turn("hello there")), /hasPriorFlag/);
  assert.throws(() => buildStudentGraph({ ...base, hasPriorFlag: "yes" } as never), /hasPriorFlag/);
  assert.throws(() => buildStudentGraph({ ...base, hasPriorFlag: undefined } as never), /hasPriorFlag/);
  assert.throws(() => buildStudentGraph({ ...base, hasPriorFlag: null } as never), /hasPriorFlag/);
  // caller-supplied false cannot hide a stored flag; caller true cannot add one
  const fu = { claims: [{ text: "Public universities in Germany charge no tuition for most programmes.", kind: "fact", sourceIds: ["t-sem-1"] }] };
  const a = await runStudentTurn({ ...base, provider: studentProvider({ kbAnswer: fu }), hasPriorFlag: async () => true } as never, turn("Is tuition free in Germany?", { priorFlagInThread: false }));
  const b = await runStudentTurn({ ...base, provider: studentProvider({ kbAnswer: fu }), hasPriorFlag: async () => false } as never, turn("Is tuition free in Germany?", { priorFlagInThread: true }));
  const c = await runStudentTurn({ ...base, provider: studentProvider({ kbAnswer: fu }), hasPriorFlag: async () => { throw new Error("db down"); } } as never, turn("Is tuition free in Germany?", { priorFlagInThread: false }));
  note(`E msg(storeTrue,callerFalse)=${JSON.stringify(a.response.message)} | msg(storeFalse,callerTrue)=${JSON.stringify(b.response.message)} | msg(storeThrows)=${JSON.stringify(c.response.message)}`);
  assert.notEqual(a.response.message, b.response.message);
  assert.equal(c.response.message, a.response.message);
});

test("F. followUp limited to one checked sentence <= 200 chars", async () => {
  const run = async (followUp: string, prior = true) => {
    const provider = studentProvider({ kbAnswer: { claims: [{ text: "Public universities in Germany charge no tuition for most programmes.", kind: "fact", sourceIds: ["t-sem-1"] }], followUp } });
    const d = { kb: fakeKb(), provider, skillsDir: SKILLS_DIR, speakers: SPEAKERS, now: () => NOW, hasPriorFlag: async () => prior };
    const out = await runStudentTurn(d, turn("Is tuition free in Germany?"));
    return (out.response.message ?? "").includes(followUp.trim());
  };
  const pad = (n: number) => "Which of the cities that you like is your first choice" + " x".repeat(0) + "?";
  const base = "Which cities do you like";
  const mk = (len: number) => (base + " ".repeat(1) + "a".repeat(Math.max(0, len - base.length - 2)) + "?").slice(0, len);
  const f200 = mk(200), f201 = mk(201);
  assert.equal(f200.length, 200); assert.equal(f201.length, 201);
  const res = {
    shown200: await run(f200), shown201: await run(f201),
    oneOk: await run("Which cities are you considering?"),
    twoSent: await run("Visit the DAAD site first and apply early. Which cities are you considering?"),
    twoQ: await run("Which cities are you considering? Which courses?"),
    newlineAdvice: await run("Apply early to every university\nWhich cities are you considering?"),
    semicolonAdvice: await run("Apply early to every university; which cities are you considering?"),
    spelledNumber: await run("Did you know it costs five thousand euros a year?"),
    spelledNumber2: await run("Would a fee of twenty thousand euros worry you?"),
    exclam: await run("Wow great! Which cities are you considering?"),
    crisisPhrase: await run("Can we keep this a secret between us?"),
    devanagariDanda: await run("आप कौन से शहर देख रहे हैं। आप कौन सा कोर्स चाहते हैं?"),
    noteQ: pad(0) && await run("Which cities are you considering?   "),
  };
  note(`F followUp shown map: ${JSON.stringify(res)}`);
  assert.equal(res.shown200, true); assert.equal(res.shown201, false);
  assert.equal(res.twoSent, false); assert.equal(res.twoQ, false); assert.equal(res.oneOk, true);
});
