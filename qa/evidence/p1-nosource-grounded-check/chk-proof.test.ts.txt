const AUTH = { role: "student" as const, studentId: "syn-stu-77", tenantId: "tenant-xyz" };
const NO_HITS = { status: "ok" as const, hits: [], dropped: 0 };
const NOT_SERVED = { status: "no_sourced_answer" as const, answer: NO_SOURCED_ANSWER_TEXT, citations: { internal: [], web: [] }, verdict: "incorrect" as const, reason: "", withheld: ["verdict_not_correct"] };
const DISC = "General guidance — I couldn't find this in Vidysea's verified sources, so please confirm fees, deadlines and amounts on the official website.";
const D110 = "Public universities in Germany usually charge no tuition, but most ask for a semester contribution of around 300 euros.";
const GTEXT = "Most German public universities charge no tuition, but expect a semester contribution of around 300 euros.";
const mk = (anonymous: boolean) => fakeKb({ search: NO_HITS, ...(anonymous ? { anonymous: true } : { ask: NOT_SERVED }) });
const guidance = (answer: unknown = { answer: D110 }) => new MockProvider({ responses: { "kb-general-guidance": answer as never } });
const gcalls = (p: MockProvider) => p.calls.filter((c) => c.skillName === "kb-general-guidance");
const auth = (text: string) => ({ mode: "authenticated" as const, auth: AUTH, threadId: "thr-secret-9", turnId: "turn-secret-9", text });

type Cand = { finishReason?: string; text?: string; chunks?: unknown[] };
function fakeFetch(c: Cand | "http500" | "throw" | "hang" | "badjson", wire: { url: string; init: RequestInit }[]): typeof fetch {
  return (async (url: string, init: RequestInit) => {
    wire.push({ url: String(url), init });
    if (c === "throw") throw new Error("ECONNRESET");
    if (c === "hang") return new Promise((_res, rej) => init.signal?.addEventListener("abort", () => rej(Object.assign(new Error("x"), { name: "AbortError" }))));
    if (c === "http500") return new Response("boom", { status: 500 });
    if (c === "badjson") return new Response("{not json", { status: 200 });
    const body = { candidates: [{ finishReason: c.finishReason ?? "STOP", content: { parts: [{ text: c.text ?? GTEXT }] }, groundingMetadata: { groundingChunks: c.chunks ?? [] } }] };
    return new Response(JSON.stringify(body), { status: 200, headers: { "content-type": "application/json" } });
  }) as unknown as typeof fetch;
}
const gem = (c: Parameters<typeof fakeFetch>[0], wire: { url: string; init: RequestInit }[], timeoutMs?: number) =>
  new GeminiGroundedAnswerer({ model: "gemini-2.5-flash", apiKey: "TESTKEY-NOT-REAL", fetchImpl: fakeFetch(c, wire), ...(timeoutMs ? { timeoutMs } : {}) });
const base = (over: Partial<StudentGraphDeps>): StudentGraphDeps => ({ kb: mk(false), provider: guidance(), skillsDir: SKILLS_DIR, speakers: SPEAKERS, now: () => NOW, hasPriorFlag: NO_PRIOR_FLAG, recordGap: async () => {}, countAnonymous: async () => {}, ...over });

const chunk = (uri: string, title = "T") => ({ web: { uri, title } });
const MANY = [
  chunk("https://a.example.test/1", "A"), chunk("http://insecure.example.test/x"), chunk("javascript:alert(1)"), chunk("https://u:p@cred.example.test/x"),
  chunk("https://b.example.test/2", "B <script>alert(1)</script>"), chunk("https://c.example.test/3"), chunk("https://d.example.test/4"), chunk("https://e.example.test/5"), chunk("https://f.example.test/6"), chunk("https://g.example.test/7"),
];

test("G1 wire: only the redacted question + fixed prompt reach Gemini; key in header only; sources https max 5; none inline", async () => {
  for (const mode of ["auth", "anon"] as const) {
    const wire: { url: string; init: RequestInit }[] = [];
    const p = guidance();
    const d = base({ kb: mk(mode === "anon"), provider: p, grounding: gem({ chunks: MANY }, wire) });
    const q = mode === "auth"
      ? "My name is Nisha Varadkar, DOB 04/05/2004, passport K1234567, phone 98765 43210, mail nisha@example.com, link https://x.example.test/p. What is the tuition at German public universities?"
      : "What did Nisha Varadkar say, see https://x.example.test/p ? What is the tuition at German public universities?";
    console.log("ANONSTATE", mode);
    const out = await runStudentTurn(d, mode === "auth" ? auth(q) : anon(q));
    assert.equal(wire.length, 1, mode);
    const w = wire[0]!;
    assert.ok(!w.url.includes("TESTKEY"), "key not in URL");
    assert.equal((w.init.headers as Record<string, string>)["x-goog-api-key"], "TESTKEY-NOT-REAL");
    const body = JSON.parse(String(w.init.body));
    const sent = JSON.stringify(body);
    for (const bad of ["Nisha", "Varadkar", "04/05/2004", "K1234567", "98765", "nisha@example.com", "x.example.test", "syn-stu-77", "tenant-xyz", "thr-secret", "turn-secret", "TESTKEY"]) assert.ok(!sent.includes(bad), `${mode}: wire leaks ${bad}`);
    assert.deepEqual(Object.keys(body).sort(), ["contents", "generationConfig", "systemInstruction", "tools"]);
    assert.equal(body.contents.length, 1);
    assert.match(body.contents[0].parts[0].text, /tuition at German public universities/);
    assert.deepEqual(body.tools, [{ google_search: {} }]);
    assert.equal(out.response.state, "no_sourced_answer");
    assert.equal(out.response.message, `${DISC} ${GTEXT}`);
    assert.ok(out.response.sources.length <= 5 && out.response.sources.length === 5);
    assert.ok(out.response.sources.every((s) => s.kind === "web" && s.url!.startsWith("https://")));
    assert.ok(!/https?:|<script/.test(out.response.message ?? ""));
    assert.ok(out.response.sources.every((s) => !/[<>]/.test(s.title ?? "")), "title sanitised");
    assert.equal(gcalls(p).length, 0);
  }
});

test("G2 fallbacks via the REAL Gemini class: flag off / 500 / throw / hang / badjson / MAX_TOKENS / empty / bad link -> D-110 text, no crash", async () => {
  const cases: [string, Parameters<typeof fakeFetch>[0] | "off"][] = [
    ["off", "off"], ["500", "http500"], ["throw", "throw"], ["hang", "hang"], ["badjson", "badjson"],
    ["max", { finishReason: "MAX_TOKENS" }], ["safety", { finishReason: "SAFETY" }], ["empty", { text: "  " }], ["link", { text: "See ucas.com/apply for fees." }], ["bare", { text: "Details are on daad.de." }],
  ];
  for (const [name, c] of cases) {
    const wire: { url: string; init: RequestInit }[] = [];
    const p = guidance();
    const d = base({ provider: p, expertTimeoutMs: 300, ...(c === "off" ? {} : { grounding: gem(c, wire, 5000) }) });
    const out = await runStudentTurn(d, auth("What is the tuition at German public universities?"));
    assert.equal(out.response.message, `${DISC} ${D110}`, name);
    assert.deepEqual(out.response.sources, [], name);
    assert.equal(out.response.state, "no_sourced_answer", name);
  }
  // both fail -> refusal
  const d = base({ provider: guidance({}), grounding: gem("http500", []) });
  assert.equal((await runStudentTurn(d, auth("What is the tuition in Germany?"))).response.message, NO_SOURCED_ANSWER_TEXT);
  const d2 = base({ kb: mk(true), provider: guidance({}), grounding: gem("throw", []) });
  assert.equal((await runStudentTurn(d2, anon("What is the tuition in Germany?"))).response.message, ANONYMOUS_NO_ANSWER_TEXT);
});

test("G5 order: anonymous screen call precedes any grounded/guidance call; possible/outage -> crisis msg w/o web; crisisGuard first", async () => {
  const order: string[] = [];
  const screenProv = (sev: string | "throw"): LlmProvider => ({ kind: "mock", model: "s", generate: async () => { order.push("screen"); if (sev === "throw") throw new Error("down"); return { severity: sev, likelyBenign: false, rationale: "x" }; } }) as unknown as LlmProvider;
  const wireFetch = (c: Cand): typeof fetch => ((async (url: string, init: RequestInit) => { order.push("web"); return fakeFetch(c, [])(url, init); }) as unknown as typeof fetch);
  const mkG = () => new GeminiGroundedAnswerer({ model: "gemini-2.5-flash", apiKey: "k", fetchImpl: wireFetch({}) });
  const text = "What is the tuition at German public universities?";
  // clear
  let d = base({ kb: mk(true), grounding: mkG(), safetyScreen: { provider: screenProv("none"), anonymous: "no_source" } });
  let out = await runStudentTurn(d, anon(text));
  assert.deepEqual(order.splice(0), ["screen", "web"]); assert.equal(out.response.message, `${DISC} ${GTEXT}`);
  // possible
  d = base({ kb: mk(true), grounding: mkG(), safetyScreen: { provider: screenProv("high"), anonymous: "no_source" } });
  out = await runStudentTurn(d, anon(text));
  assert.deepEqual(order.splice(0), ["screen"]); assert.equal(out.response.state, "escalated"); assert.equal(out.response.message, ANONYMOUS_CRISIS_MESSAGE);
  // outage
  d = base({ kb: mk(true), grounding: mkG(), safetyScreen: { provider: screenProv("throw"), anonymous: "no_source" }, onWriteError: () => {} });
  out = await runStudentTurn(d, anon(text));
  assert.deepEqual(order.splice(0), ["screen"]); assert.equal(out.response.state, "escalated"); assert.equal(out.response.message, ANONYMOUS_CRISIS_MESSAGE);
  // anonymous:true and authenticated
  for (const [kbAnon, mode, cfg] of [[true, "anon", { anonymous: true }], [false, "auth", { authenticated: true }]] as const) {
    d = base({ kb: mk(kbAnon), grounding: mkG(), safetyScreen: { provider: screenProv("none"), ...cfg } });
    out = await runStudentTurn(d, mode === "anon" ? anon(text) : auth(text));
    const o = order.splice(0); assert.ok(o.indexOf("screen") !== -1 && o.indexOf("screen") < o.indexOf("web"), `${mode}: ${o}`);
    d = base({ kb: mk(kbAnon), grounding: mkG(), safetyScreen: { provider: screenProv("high"), ...cfg } });
    out = await runStudentTurn(d, mode === "anon" ? anon(text) : auth(text));
    assert.ok(!order.splice(0).includes("web"), `${mode}: web call after possible`); assert.equal(out.response.state, "escalated");
  }
  // crisis first
  d = base({ kb: mk(true), grounding: mkG(), safetyScreen: { provider: screenProv("none"), anonymous: "no_source" } });
  out = await runStudentTurn(d, anon("I want to kill myself, I don't want to live anymore"));
  assert.deepEqual(order.splice(0), []); assert.equal(out.response.state, "escalated");
});
