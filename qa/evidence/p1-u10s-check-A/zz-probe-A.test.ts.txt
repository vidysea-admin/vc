import assert from "node:assert/strict";
import test from "node:test";
import { VOICE_CRISIS_NO_WORDS_TEXT } from "../src/voice.js";
import { CRISIS_AUDIO, CRISIS_PHRASE, CRISIS_TEXT_FIXED, adult, minorNoConsent, newThread, studentToken, until, voiceWorld, type VoiceWorld } from "./voice-fixtures.js";
import { CRISIS_RESUME_GAP_MS_DEFAULT, type LiveCrisisDetector } from "@vc/providers";

const speak = (w: VoiceWorld, text: string) => w.provider.backends[0]!.emitInputTranscript(text, true);
const gap = () => new Promise((r) => setTimeout(r, CRISIS_RESUME_GAP_MS_DEFAULT + 300));
const detectEmptyFires: LiveCrisisDetector = (text) => (text.trim() === "" || text.includes(CRISIS_PHRASE) ? { crisis: true, level: "crisis" } : { crisis: false });
const throwingNames = () => ({ profileNames: async () => [123 as unknown as string] });

async function run(claims: any, extra: Record<string, unknown>, body: (w: VoiceWorld, esc: any[]) => Promise<void>) {
  const esc: any[] = [];
  const w = await voiceWorld({ deps: { crisis: { message: { text: CRISIS_TEXT_FIXED, audio: CRISIS_AUDIO }, detect: detectEmptyFires }, escalateForTenant: () => async (i: any) => void esc.push(i), ...extra } as any });
  try {
    const threadId = await newThread(w.chat, await studentToken(adult()));
    await w.open(claims, threadId);
    await body(w, esc);
  } finally { await w.cleanup(); }
}
const flagIds = (w: VoiceWorld) => w.fakeDb.collection<any>("student_flags").all().map((f: any) => f.turnId);

test("P-H3a: three no-words crises -> 3 escalations, 3 flags, 3 distinct ids", async () => {
  await run(adult(), {}, async (w, esc) => {
    for (let i = 1; i <= 3; i++) { speak(w, "   "); await until(() => esc.length === i, `esc ${i}`); w.provider.backends[0]!.emitTurnComplete(); await gap(); }
    await w.voice.settled();
    const ids = esc.map((e) => e.turns[0].turnId);
    assert.equal(new Set(ids).size, 3, JSON.stringify(ids));
    assert.equal(new Set(flagIds(w)).size, 3, JSON.stringify(flagIds(w)));
    assert.deepEqual([...ids].sort(), [...flagIds(w)].sort(), "flag ids == escalation ids");
    for (const e of esc) assert.equal(e.turns[0].text, VOICE_CRISIS_NO_WORDS_TEXT);
  });
});

test("P-H3b: two crises WITH words -> 2 flags/2 escalations, distinct ids; ids are stored voice turn ids", async () => {
  await run(adult(), {}, async (w, esc) => {
    speak(w, `${CRISIS_PHRASE} one`); await until(() => esc.length === 1, "esc1"); w.provider.backends[0]!.emitTurnComplete(); await gap();
    speak(w, `${CRISIS_PHRASE} two`); await until(() => esc.length === 2, "esc2");
    await w.voice.settled();
    const ids = esc.map((e) => e.turns[0].turnId);
    console.error("PROBE H3b ids", JSON.stringify(ids), "flags", JSON.stringify(flagIds(w)));
    assert.notEqual(ids[0], ids[1]);
    assert.equal(w.fakeDb.collection<any>("student_flags").all().length, 2);
    assert.match(esc[0].turns[0].text, /one/); assert.match(esc[1].turns[0].text, /two/);
  });
});

test("P-H3c: mixed no-words / words / no-words -> three distinct ids and flags", async () => {
  await run(adult(), {}, async (w, esc) => {
    speak(w, "   "); await until(() => esc.length === 1, "e1"); w.provider.backends[0]!.emitTurnComplete(); await gap();
    speak(w, `${CRISIS_PHRASE} mid`); await until(() => esc.length === 2, "e2"); w.provider.backends[0]!.emitTurnComplete(); await gap();
    speak(w, "   "); await until(() => esc.length === 3, "e3");
    await w.voice.settled();
    const ids = esc.map((e) => e.turns[0].turnId);
    console.error("PROBE H3c ids", JSON.stringify(ids));
    assert.equal(new Set(ids).size, 3); assert.equal(new Set(flagIds(w)).size, 3);
  });
});

test("P-H3d: unconsented minor, two crises with words -> 2 distinct flags, zero escalations, no text in logs", async () => {
  await run(minorNoConsent(), {}, async (w, esc) => {
    speak(w, `${CRISIS_PHRASE} minorsecret`); await until(() => w.fakeDb.collection<any>("student_flags").all().length === 1, "f1"); w.provider.backends[0]!.emitTurnComplete(); await gap();
    speak(w, `${CRISIS_PHRASE} minorsecret2`); await until(() => w.fakeDb.collection<any>("student_flags").all().length === 2, "f2");
    await w.voice.settled();
    assert.equal(esc.length, 0);
    assert.equal(new Set(flagIds(w)).size, 2, JSON.stringify(flagIds(w)));
    assert.ok(!JSON.stringify(w.logs).includes("minorsecret"));
  });
});

test("P-H1a: redaction throws on two crises with words -> 2 flags, 2 escalations (marker), 2 redact_failed log lines, no raw words anywhere", async () => {
  await run(adult(), throwingNames(), async (w, esc) => {
    speak(w, `${CRISIS_PHRASE} rawsecret-A`); await until(() => esc.length === 1, "e1"); w.provider.backends[0]!.emitTurnComplete(); await gap();
    speak(w, `${CRISIS_PHRASE} rawsecret-B`); await until(() => esc.length === 2, "e2");
    await w.voice.settled();
    for (const e of esc) { assert.equal(e.turns[0].text, VOICE_CRISIS_NO_WORDS_TEXT); assert.equal(e.turns[0].translatedText, VOICE_CRISIS_NO_WORDS_TEXT); }
    assert.notEqual(esc[0].turns[0].turnId, esc[1].turns[0].turnId);
    assert.equal(w.fakeDb.collection<any>("student_flags").all().length, 2);
    const failed = w.logs.filter((l: any) => l.event === "voice_crisis_redact_failed");
    assert.equal(failed.length, 2);
    console.error("PROBE H1a log line", JSON.stringify(failed[0]));
    const all = JSON.stringify(w.logs) + JSON.stringify(esc);
    assert.ok(!all.includes("rawsecret") && !all.includes("kill myself"), "no raw words in logs or escalations");
  });
});

test("P-H1b: redaction throws, then recovers conceptually: no-words crisis after a throw still escalates with its own id", async () => {
  await run(adult(), throwingNames(), async (w, esc) => {
    speak(w, `${CRISIS_PHRASE} rawsecret-C`); await until(() => esc.length === 1, "e1"); w.provider.backends[0]!.emitTurnComplete(); await gap();
    speak(w, "   "); await until(() => esc.length === 2, "e2");
    await w.voice.settled();
    assert.notEqual(esc[0].turns[0].turnId, esc[1].turns[0].turnId);
    assert.equal(w.fakeDb.collection<any>("student_flags").all().length, 2);
  });
});
