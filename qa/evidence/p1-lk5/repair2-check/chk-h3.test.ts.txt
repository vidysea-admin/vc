function cls(m: string): string {
  if (m.includes("Sorry, I couldn't")) return "APOLOGY";
  if (m.includes("I don't have a sourced answer")) return "NOANSWER";
  if (m.startsWith(STUDENT_CRISIS_MESSAGE.slice(0, 40))) return "CRISIS";
  if (m.includes("Public universities in Germany charge") || m.includes("German public universities charge") || m.includes("semester contribution of about 300 euros") || m.includes("Expect ")) return "ENGLISH";
  return "LOCALIZED";
}

async function drive(label: string, o: { det?: typeof HI; q?: string; kbAnswer: unknown; flag?: boolean; search?: SearchToolOutput }) {
  const provider = studentProvider({ kbAnswer: o.kbAnswer });
  const deps: StudentGraphDeps = {
    kb: fakeKb(o.search ? { search: o.search } : {}), provider, skillsDir: SKILLS_DIR, speakers: SPEAKERS, now: () => NOW,
    hasPriorFlag: o.flag ? async () => true : NO_PRIOR_FLAG,
    escalate: async () => {}, logFlag: async () => {}, countAnonymous: async () => {},
    detectLanguage: o.det ?? HI,
  };
  const out = await runStudentTurn(deps, turn(o.q ?? HI_Q));
  const m = out.response.message ?? "";
  console.log(`CASE ${label} => ${cls(m)} | ${m.slice(0, 160).replace(/\n/g, " ")}`);
  return { m, c: cls(m) };
}
