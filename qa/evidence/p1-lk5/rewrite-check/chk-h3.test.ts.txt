const AUTH = { role: "student" as const, studentId: "syn-chk", tenantId: "sandbox" };
const turn = (text: string) => ({ mode: "authenticated" as const, auth: AUTH, threadId: "th-c", turnId: "tu-c", text });
const det = (lang: string, script: DetectedLanguage["script"]) => async (): Promise<DetectedLanguage> => ({ lang, script, confidence: 0.9, source: "model" });
const HI = det("hi", "devanagari"), HING = det("hi-Latn", "latin");
function run(claims: unknown[], replyLang: "hi" | "hi-Latn", followUpLocalized?: string, followUp?: string) {
  const provider = studentProvider({ kbAnswer: { replyLang, claims, ...(followUp ? { followUp } : {}), ...(followUpLocalized ? { followUpLocalized } : {}) } });
  const d: StudentGraphDeps = {
    kb: fakeKb(), provider, skillsDir: SKILLS_DIR, speakers: SPEAKERS, now: () => NOW, hasPriorFlag: NO_PRIOR_FLAG,
    escalate: async () => {}, logFlag: async () => {}, countAnonymous: async () => {}, detectLanguage: replyLang === "hi" ? HI : HING,
  };
  return runStudentTurn(d, turn(replyLang === "hi" ? "फीस कितनी है" : "fees kitni hai"));
}

test("A english factual -> english wording via runStudentTurn", async () => {
  for (const [en, lo, rl] of A) {
    const out = await run([{ text: en, localized: lo, kind: "fact", sourceIds: ["t-sem-1"] }], rl);
    const m = out.response.message ?? "";
    const ok = m.includes(en) && !m.includes(lo);
    rows.push(`A ${ok ? "OK " : "BAD"} | ${lo} | ${m.slice(0, 90).replace(/\n/g, "/")}`);
  }
  assert.ok(rows.filter((r) => r.startsWith("A BAD")).length === 0, rows.filter((r) => r.startsWith("A BAD")).join("\n"));
});

// D: mixed + alignment + follow-up
test("D mixed claim, misalignment, follow-up", async () => {
  const en = `${FREE_EN} ${NUM_EN} Good luck with your plans!`;
  const lo = `${FREE_HI} सेमेस्टर शुल्क करीब तीन सौ यूरो है। आपकी योजनाओं के लिए शुभकामनाएँ!`;
  const r = localizedClaimText(en, lo, "hi", []);
  assert.equal(r, `${FREE_HI} ${NUM_EN} आपकी योजनाओं के लिए शुभकामनाएँ!`);
  rows.push(`D mixed OK | ${r}`);
  assert.equal(localizedClaimText(`${FREE_EN} ${NUM_EN}`, "सरकारी विश्वविद्यालयों में अधिकतर कोर्स मुफ़्त हैं।", "hi", []), undefined);
  assert.equal(localizedClaimText(FREE_EN, "सरकारी कोर्स मुफ़्त हैं। सेमेस्टर शुल्क सत्तर है।", "hi", []), undefined);
  const out = await run([{ text: FREE_EN, localized: "सरकारी कोर्स मुफ़्त हैं। सेमेस्टर शुल्क सत्तर है।", kind: "fact", sourceIds: ["t-sem-1"] }], "hi");
  assert.ok(out.response.message!.includes(FREE_EN) && !out.response.message!.includes("सत्तर"), out.response.message);
  const out2 = await run([
    { text: FREE_EN, localized: FREE_HI, kind: "fact", sourceIds: ["t-sem-1"] },
    { text: NUM_EN, localized: "सेमेस्टर शुल्क। करीब तीन सौ यूरो।", kind: "number", sourceIds: ["t-sem-1"] },
  ], "hi");
  const m2 = out2.response.message!;
  rows.push(`D twoclaim-misaligned | ${m2.slice(0, 200).replace(/\n/g, "/")}`);
  assert.ok(m2.includes(FREE_EN) && m2.includes(NUM_EN) && !m2.includes(FREE_HI), m2);
  const f1 = await run([{ text: FREE_EN, localized: FREE_HI, kind: "fact", sourceIds: ["t-sem-1"] }], "hi", "आप कौन सा विषय पढ़ना चाहते हैं?", "Which subject would you like to study?");
  rows.push(`D followup-free | ${f1.response.message!.replace(/\n/g, "/")}`);
  assert.ok(f1.response.message!.includes("आप कौन सा विषय"), f1.response.message);
  const f2 = await run([{ text: FREE_EN, localized: FREE_HI, kind: "fact", sourceIds: ["t-sem-1"] }], "hi", "आप 2027 में आवेदन करेंगे?", "Which subject would you like to study?");
  rows.push(`D followup-digit | ${f2.response.message!.replace(/\n/g, "/")}`);
  assert.ok(!f2.response.message!.includes("2027"));
});
