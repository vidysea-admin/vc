async function post(path: string, raw: string | object, tok?: string) {
  const res = await fetch(`${base}${path}`, {
    method: "POST",
    headers: { "content-type": "application/json", ...(tok ? { authorization: `Bearer ${tok}` } : {}) },
    body: typeof raw === "string" ? raw : JSON.stringify(raw),
  });
  const body = await res.text();
  let json: any;
  try { json = JSON.parse(body); } catch { json = undefined; }
  return { status: res.status, body, json };
}
async function thread(): Promise<string> {
  const r = await post(threadsPath("stu-1"), {}, token);
  assert.equal(r.status, 201, r.body);
  return r.json.threadId as string;
}

  ["clientTurnId space", JSON.stringify({ text, clientTurnId: `a ${MARK}` })],

  ["clientTurnId number", JSON.stringify({ text, clientTurnId: 987654 })],
  ["clientTurnId null", JSON.stringify({ text, clientTurnId: null })],
  ["clientTurnId empty", JSON.stringify({ text, clientTurnId: "" })],
  ["both", JSON.stringify({ text, extra: MARK, clientTurnId: `../${MARK}` })],
];
const snapshot = () => everythingStored(cw.fakeDb) + JSON.stringify(cw.escalations) + JSON.stringify(cw.logs);

test("AUTH crisis x malformed: escalated crisis answer with 14416, nothing malformed persisted", async () => {
  for (const [script, text] of CRISIS) {
    for (const [name, raw] of bad(text)) {
      const e0 = cw.escalations.length;
      const f0 = cw.fakeDb.collection(COLLECTION_NAMES.student_flags).all().length;
      const t = await thread(); // separate thread => separate slot
      const r = await post(turnsPath("stu-1", t), raw, token);
      assert.equal(r.status, 200, `${script}/${name}: ${r.body}`);
      assert.equal(r.json.state, "escalated", `${script}/${name}`);
      assert.match(r.json.message, /14416/, `${script}/${name}`);
      assert.equal(cw.escalations.length - e0, 1, `${script}/${name}: one escalation`);
      assert.equal(cw.fakeDb.collection(COLLECTION_NAMES.student_flags).all().length - f0, 1, `${script}/${name}: one flag`);
      assert.ok(!r.body.includes(MARK) && !r.body.includes("987654"), `${script}/${name}: not echoed`);
      if (r.json.turnId !== undefined) assert.match(r.json.turnId, /^[A-Za-z0-9][A-Za-z0-9._-]{0,127}$/, `${script}/${name}: turnId sane`);
    }
  }
  const all = snapshot();
  assert.ok(!all.includes(MARK), "marker never stored/logged/escalated");
  assert.ok(!all.includes("987654"), "numeric id never stored");
  assert.ok(!all.includes("xxxxxxxxxx"), "129-char id never stored");
});

test("AUTH rate-limited student + bad field + crisis -> crisis answer; benign still 429", async () => {
  const w2 = chatWorld();
  const s2 = createApiServer(w2.handler);
  await new Promise<void>((r) => s2.listen(0, "127.0.0.1", r));
  const b2 = `http://127.0.0.1:${(s2.address() as { port: number }).port}`;
  try {
    const tok = await studentToken(adult());
    const mk = async (p: string, method: string, body?: string) => { const r = await fetch(b2 + p, { method, headers: { authorization: `Bearer ${tok}`, "content-type": "application/json" }, ...(body !== undefined ? { body } : {}) }); return { status: r.status, text: await r.text() }; };
    const tid = JSON.parse((await mk(threadsPath("stu-1"), "POST", "{}")).text).threadId as string;
    let limited = 0;
    for (let i = 0; i < 60 && !limited; i++) { const r = await mk(`${threadsPath("stu-1")}/${tid}`, "GET"); if (r.status === 429) limited = i + 1; }
    assert.ok(limited > 0, "student reached the rate limit");
    assert.equal((await mk(turnsPath("stu-1", tid), "POST", JSON.stringify({ text: GERMANY }))).status, 429);
    for (const [script, text] of CRISIS) {
      for (const [name, raw] of bad(text)) {
        const r = await mk(turnsPath("stu-1", tid), "POST", raw);
        assert.equal(r.status, 200, `${script}/${name}: ${r.text}`);
        assert.equal(JSON.parse(r.text).state, "escalated");
        assert.match(JSON.parse(r.text).message, /14416/);
      }
    }
    // benign + bad field while limited: still a 400 or 429 (never a crisis path); must not be 200
    const rb = await mk(turnsPath("stu-1", tid), "POST", JSON.stringify({ text: GERMANY, role: MARK }));
    assert.ok(rb.status === 400 || rb.status === 429, `benign malformed while limited -> ${rb.status}`);
    assert.equal(w2.provider.calls.length, 0);
  } finally {
    await new Promise<void>((r) => s2.close(() => r()));
    w2.cleanup();
  }
});

test("AUTH replay: a valid clientTurnId replayed after a crisis-with-bad-field turn", async () => {
  const w = chatWorld();
  const s = createApiServer(w.handler);
  await new Promise<void>((r) => s.listen(0, "127.0.0.1", r));
  const b = `http://127.0.0.1:${(s.address() as { port: number }).port}`;
  try {
    const tok = await studentToken(adult());
    const mk = async (p: string, body?: object | string) => { const r = await fetch(b + p, { method: "POST", headers: { authorization: `Bearer ${tok}`, "content-type": "application/json" }, body: typeof body === "string" ? body : JSON.stringify(body ?? {}) }); const text = await r.text(); return { status: r.status, text, json: JSON.parse(text) }; };
    const tid = (await mk(threadsPath("stu-1"))).json.threadId as string;
    const path = turnsPath("stu-1", tid);
    const CR = "I want to kill myself";
    const first = await mk(path, { text: CR, clientTurnId: "../evil", role: MARK });
    assert.equal(first.status, 200);
    assert.equal(first.json.state, "escalated");
    assert.notEqual(first.json.turnId, "../evil");
    const e1 = w.escalations.length;
    assert.equal(e1, 1);
    // now a valid id
    const v1 = await mk(path, { text: CR, clientTurnId: "valid-replay-1" });
    assert.equal(v1.status, 200);
    assert.equal(v1.json.state, "escalated");
    const e2 = w.escalations.length;
    const v2 = await mk(path, { text: CR, clientTurnId: "valid-replay-1" });
    assert.equal(v2.status, 200);
    assert.equal(v2.json.state, "escalated");
    assert.equal(v2.json.turnId, v1.json.turnId, "replay returns the same turn id");
    assert.equal(w.escalations.length, e2, "replay does not escalate again");
    // valid id + unknown key keeps the id (idempotent) and replays too
    const v3 = await mk(path, { text: CR, clientTurnId: "valid-replay-1", role: MARK });
    assert.equal(v3.status, 200);
    assert.equal(v3.json.turnId, "valid-replay-1");
    assert.equal(w.escalations.length, e2);
    const all = everythingStored(w.fakeDb) + JSON.stringify(w.escalations) + JSON.stringify(w.logs);
    assert.ok(!all.includes(MARK) && !all.includes("../evil"), "malformed fields never kept");
  } finally {
    await new Promise<void>((r) => s.close(() => r()));
    w.cleanup();
  }
});

// ---- scan counting, independent instrument: NFKD normalisations whose receiver is the exact text (3 per crisisGuard scan) ----
async function nfkd(text: string, fn: () => Promise<unknown>): Promise<number> {
  const orig = String.prototype.normalize;
  let n = 0;
  String.prototype.normalize = function (this: string, form?: string): string {
    if (form === "NFKD" && String(this) === text) n++;
    return orig.call(this, form as never);
  } as typeof String.prototype.normalize;
  try { await fn(); } finally { String.prototype.normalize = orig; }
  return n;
}

test("SCAN: malformed crisis == well-formed crisis; malformed benign == one scan (both routes, independent instrument)", async () => {
  const CT = "Chk scan probe alpha: I want to kill myself";
  const BT = "Chk scan probe beta: what are tuition fees in Germany?";
  // auth
  const t1 = await thread(), t2 = await thread(), t3 = await thread();
  const wf = await nfkd(CT, () => post(turnsPath("stu-1", t1), { text: CT }, token));
  const mf = await nfkd(CT, () => post(turnsPath("stu-1", t2), { text: CT, role: MARK }, token));
  const t4 = await thread();
  const mf2 = await nfkd(CT, () => post(turnsPath("stu-1", t4), { text: CT, clientTurnId: "../x", role: MARK }, token));
  const bm = await nfkd(BT, () => post(turnsPath("stu-1", t3), { text: BT, role: MARK }, token));
  const direct = await nfkd(BT, async () => { const { crisisGuard } = await import("@vc/engine"); crisisGuard(BT); });
  console.log("AUTH scan NFKD counts: wellformed", wf, "malformed", mf, "both-bad", mf2, "benign-malformed", bm, "direct-one-scan", direct);
  assert.ok(wf >= 3);
  assert.equal(mf, wf);
  assert.equal(mf2, wf);
  assert.equal(bm, direct);
  // anon
  const awf = await nfkd(CT, () => post(ANONYMOUS_PATH, { text: CT }));
  const amf = await nfkd(CT, () => post(ANONYMOUS_PATH, { text: CT, role: MARK }));
  const abm = await nfkd(BT, () => post(ANONYMOUS_PATH, { text: BT, role: MARK }));
  const abw = await nfkd(BT, () => post(ANONYMOUS_PATH, { text: BT }));
  console.log("ANON scan NFKD counts: wellformed", awf, "malformed", amf, "benign-malformed", abm, "benign-wellformed", abw);
  assert.equal(amf, awf);
  assert.equal(abm, direct);
});
