def run(cmd_dir, files):
    p = subprocess.run(["pnpm", "-C", cmd_dir, "exec", "tsx", "--test", "--test-timeout=30000"] + files, capture_output=True, text=True, encoding="utf-8", errors="replace", shell=True)
    out = p.stdout + p.stderr
    t = re.search(r"ℹ tests (\d+)", out); f = re.search(r"ℹ fail (\d+)", out)
    names = re.findall(r"^✖ (.*?) \(", out, re.M)
    return (int(t.group(1)) if t else -1, int(f.group(1)) if f else -1, sorted(set(names))[:4])

def read(path):
    with open(os.path.join(ROOT, path), "rb") as fh:
        return fh.read().decode("utf-8")

def main():
    which = sys.argv[1:] or None
    for fid, crit, path, old, new in M:
        if which and fid not in which: continue
        orig = read(path)
        crlf = "\r\n" in orig
        o = old
        if o not in orig:
            print(fid, "PATTERN NOT FOUND"); continue
        assert orig.count(o) == 1, (fid, orig.count(o))
        with open(os.path.join(ROOT, path), "wb") as fh:
            fh.write(orig.replace(o, new).encode("utf-8"))
        try:
            e = run(ROOT + "/packages/engine", ENG)
            a = run(ROOT + "/apps/api", API) if fid == "F-L2" or fid == "F-L3" else None
            print(fid, crit, "engine tests/fail", e[0], e[1], e[2], "api", a, flush=True)
        finally:
            with open(os.path.join(ROOT, path), "wb") as fh:
                fh.write(orig.encode("utf-8"))
    r = subprocess.run(["git", "-C", ROOT, "status", "--short"], capture_output=True, text=True)
    print("STATUS after:", r.stdout)

if __name__ == "__main__":
    if sys.argv[1:] == ["baseline"]:
        print("baseline engine", run(ROOT + "/packages/engine", ENG), "api", run(ROOT + "/apps/api", API))
    else:
        main()
